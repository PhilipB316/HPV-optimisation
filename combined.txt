import numpy as np
from scipy import stats
from scipy.optimize import curve_fit

class Model:
    p_max = 2000       # realised power is clipped to this
    valid_from = 0     # shortest duration (s) the fitted curve is valid for

    def __init__(self, **params):
        self.params = params

    @property
    def cp(self):
        return self.params["CP"]


class CPModel(Model):
    """Model 1: critical power and W' (P = CP + W'/t).

    params: CP (W), W_prime (J). Fitted to the 60-1800 s part of the MMP
    data. Power is capped at 1000 W. W' recovers below CP using the
    Bartram et al. time constant.
    """
    p_max = 1000
    valid_from = 60

    def fit(self, t, p):
        """Linearised hyperbolic fit: P = W'/t + CP, using 60-1800 s only."""
        mask = (t >= 60) & (t <= 1800)
        W_prime, CP, rvalue, _, _ = stats.linregress(1.0 / t[mask], p[mask])
        self.params = {"CP": CP, "W_prime": W_prime, "r": rvalue}
        return self

    def predict(self, t):
        return self.params["CP"] + self.params["W_prime"] / t

    def realise(self, power_profile, dt):
        """
        Depletion (P > CP):  dW'/dt = -(P - CP)
        Recovery  (P < CP):  dW'/dt = (W' - W'_bal) / tau_W'
        """
        CP, W_prime = self.params["CP"], self.params["W_prime"]
        w_bal = W_prime
        realised = np.empty(len(power_profile))
        for i, p_raw in enumerate(power_profile):
            p = min(max(p_raw, 0), self.p_max)

            if p > CP:
                wbal_consumed = min((p - CP) * dt, w_bal)
                p = CP + wbal_consumed / dt
                w_bal -= wbal_consumed
            elif p < CP:
                tau_w = 2287.2 * (CP - p) ** -0.688
                w_bal += (W_prime - w_bal) * (1.0 - np.exp(-dt / tau_w))

            realised[i] = p
        return realised


class ExpModel(Model):
    """Model 2: single-exponential power ceiling (P = (Pmax - CP)exp(-t/tau) + CP).

    params: Pmax (W), tau (s), CP (W). Recovery uses tau_rec = 3 tau, which is
    a placeholder.
    """
    def fit(self, t, p):
        def model(t, Pmax, tau, CP):
            return (Pmax - CP) * np.exp(-t / tau) + CP

        CP0 = p[-1] * 0.95
        A0 = p[0] - CP0
        tau0 = t[len(t) // 2]
        popt, _ = curve_fit(
            model, t, p, p0=(A0 + CP0, tau0, CP0),
            bounds=([0, 1, 0], [np.inf, np.inf, np.inf]),
            maxfev=10000,
        )
        self.params = dict(zip(["Pmax", "tau", "CP"], popt))
        return self

    def predict(self, t):
        Pmax, tau, CP = (self.params[k] for k in ("Pmax", "tau", "CP"))
        return (Pmax - CP) * np.exp(-t / tau) + CP

    def realise(self, power_profile, dt):
        """
        State: s in [0, 1] (freshness; 1 = fully fresh).
        Ceiling:    P_ceil(s) = (Pmax - CP) * s + CP
        Depletion (P > CP):  ds/dt = -(P - CP) / ((Pmax - CP) * tau)
        Recovery  (P <= CP): ds/dt = (1 - s) / tau_rec,  tau_rec = 3 * tau (placeholder)
        """
        Pmax, tau, CP = (self.params[k] for k in ("Pmax", "tau", "CP"))
        tau_rec = tau * 3.0
        s = 1.0
        realised = np.empty(len(power_profile))
        amplitude = Pmax - CP
        for i, p_raw in enumerate(power_profile):
            p = float(np.clip(p_raw, 0, self.p_max))
            p_ceil = amplitude * s + CP
            p = min(p, p_ceil)
            realised[i] = p

            if p > CP:
                ds = -(p - CP) / (amplitude * tau) * dt
            else:
                ds = (1.0 - s) / tau_rec * dt
            s = float(np.clip(s + ds, 0.0, 1.0))
        return realised


class BiExpModel(Model):
    """Model 3: bi-exponential power ceiling.

    P = A1 exp(-t/tau1) + A2 exp(-t/tau2) + CP.
    params: A1, tau1, A2, tau2, CP. Fast and slow components deplete above CP
    and do not recover.
    """

    def fit(self, t, p):
        def model(t, A1, tau1, A2, tau2, CP):
            return A1 * np.exp(-t / tau1) + \
                   A2 * np.exp(-t / tau2) + CP

        CP0 = p[-1] * 0.95
        A1_0 = p[0] - p[len(p) // 3]
        A2_0 = p[len(p) // 3] - CP0
        popt, _ = curve_fit(
            model, t, p, p0=(A1_0, 15, A2_0, 300, CP0),
            bounds=([0, 1, 0, 50, 0], [np.inf, 60, np.inf, 2000, np.inf]),
            maxfev=20000,
        )
        self.params = dict(zip(["A1", "tau1", "A2", "tau2", "CP"], popt))
        return self

    def predict(self, t):
        A1, tau1, A2, tau2, CP = (self.params[k] for k in ("A1", "tau1", "A2", "tau2", "CP"))
        return A1 * np.exp(-t / tau1) + A2 * np.exp(-t / tau2) + CP

    def realise(self, power_profile, dt):
        """
        Ceiling: P_ceil(s1, s2) = CP + A1*s1 + A2*s2
        Above CP each tank depletes in proportion to its share of the ceiling.
        There is no recovery.
        """
        A1, tau1, A2, tau2, CP = (self.params[k] for k in ("A1", "tau1", "A2", "tau2", "CP"))
        s1, s2 = 1.0, 1.0
        realised = np.empty(len(power_profile))

        for i, p_raw in enumerate(power_profile):
            p = float(np.clip(p_raw, 0, self.p_max))
            p_ceil = CP + A1 * s1 + A2 * s2
            p = min(p, p_ceil)
            realised[i] = p

            if p > CP:
                excess = p - CP
                denom = A1 * s1 + A2 * s2
                frac1 = (A1 * s1) / denom if denom > 1e-9 else 0.5
                ds1 = -(frac1 * excess) / (A1 * tau1) * dt
                ds2 = -((1 - frac1) * excess) / (A2 * tau2) * dt
            else:
                ds1 = 0
                ds2 = 0

            s1 = float(np.clip(s1 + ds1, 0.0, 1.0))
            s2 = float(np.clip(s2 + ds2, 0.0, 1.0))
        return realised


class DisproportionateModel(CPModel):
    """Model 4: disproportionate W' expenditure at high power (Fitton et al., Model III).

    Uses the CP and W' fitted for Model 1. Pmax is `p_max` (1000 W). W' does
    not recover.
    """
    p_max = 1000

    def realise(self, power_profile, dt):
        CP, W_prime = self.params["CP"], self.params["W_prime"]
        w_bal = W_prime
        realised = np.empty(len(power_profile))

        for i, p_raw in enumerate(power_profile):
            p = min(max(p_raw, 0), self.p_max)

            scale = (self.p_max - CP) / (self.p_max - p) if p < self.p_max else np.inf

            if p > CP:
                wbal_consumed = min((p - CP) * scale * dt, w_bal)
                p = CP + wbal_consumed / (scale * dt)
                w_bal -= wbal_consumed
            realised[i] = p
        return realised

MODELS = {
    "cp": CPModel,
    "exp": ExpModel,
    "biexp": BiExpModel,
    "disproportionate": DisproportionateModel,
}

import cma
import numpy as np
from .run import Run
from .vehicle import fastest_200m_segment

class OptimisationResult:
    """The optimised power profile and the speed it achieves.

    Attributes:
        run: Run with the optimised `watts` and simulated `calc_speeds`.
        speed: fastest 200 m average speed in km/h.
    """
    def __init__(self, run, speed):
        self.run = run # Run with the optimised watts and calc_speeds
        self.speed = speed  # fastest 200 m average speed, km/h

    def plot(self, **kwargs):
        """Plot the optimised run. Arguments are passed to `hpvopt.plot`."""
        return self.run.plot(**kwargs)

    def save(self, path):
        """Save the optimised run to a .npz file."""
        self.run.save(path)

    @classmethod
    def load(cls, path, vehicle):
        """Load a profile saved with `save`.

        The speed is recomputed for `vehicle`, so loading with a different
        vehicle gives that vehicle's speed for the same power profile.

        Args:
            path: .npz file.
            vehicle: Vehicle used to compute `speed`.
        """
        run = Run(path, label="Optimised")
        dt = run.times[1] - run.times[0]
        t, d = Optimiser(vehicle, None, dt=dt).distance_profile(run.watts)
        return cls(run, fastest_200m_segment(t, d)[0] * 3.6)

class Optimiser:
    """Finds the power profile that maximises the fastest 200 m average speed.

    The profile is searched with CMA-ES, with realised power limited by the
    physiological model.

    Args:
        vehicle: Vehicle to simulate.
        model: a fitted model from `Rider.models`.
        duration: length of the profile in s.
        dt: time step in s.
        bounds: (min, max) power in W for the search.
        roughness_weighting: penalty on power changes between steps.
        time_weighting: penalty on power drawn late in the run.
        road_length_limit: stop the simulation after this many m, or None.
    """
    def __init__(self, vehicle, model, duration=600, dt=2, bounds=(0, 2000),
                 roughness_weighting=3e-7, time_weighting=3e-5,
                 road_length_limit=None):
        self.vehicle = vehicle
        self.model = model
        self.duration = duration
        self.dt = dt
        self.bounds = bounds
        self.roughness_weighting = roughness_weighting
        self.time_weighting = time_weighting
        self.road_length_limit = road_length_limit  # m, None to disable
        self.n = int(duration / dt)
        self.time_weights = np.linspace(0.0, 1.0, self.n)

    def penalties(self, power_profile, max_idx):
        roughness = np.sum(np.diff(power_profile[:max_idx]) ** 2)
        roughness_p = roughness * self.roughness_weighting

        time_p = np.sum(self.time_weights[:max_idx] *
                        power_profile[:max_idx]) * self.time_weighting

        return roughness_p + time_p

    def distance_profile(self, realised):
        vehicle = self.vehicle
        current_speed = 0.0
        current_energy = 0.0
        current_distance = 0.0
        p_prev = realised[0]
        times = [0.0]
        distances = [0.0]
        index = 0
        for p in realised:
            index += 1
            step_energy_produced = 0.5 * (p_prev + p) * \
                self.dt * vehicle.drive_train_efficiency
            step_energy_consumed = vehicle.total_power_consumed(current_speed) * self.dt
            current_energy += (step_energy_produced - step_energy_consumed)
            if current_energy < 0:
                current_energy = 0
            prev_speed = current_speed
            current_speed = vehicle.current_speed_from_energy(current_energy)

            # advance cumulative distance (trapezoidal, km/h -> m/s)
            current_distance += 0.5 * (prev_speed + current_speed) / 3.6 * self.dt
            times.append(index * self.dt)
            distances.append(current_distance)
            p_prev = p

            if self.road_length_limit is not None and current_distance >= self.road_length_limit:
                break
        return times, distances

    def evaluate(self, power_profile):
        """
        Fitness function: negative of the fastest average speed (km/h) over any
        200m segment, plus roughness/time penalties on the portion of the power
        profile used to reach the end of that segment.
        """
        realised = self.model.realise(power_profile, self.dt)
        times, distances = self.distance_profile(realised)

        best_avg_speed_m_s, _, best_end_time = fastest_200m_segment(times, distances)
        best_speed = best_avg_speed_m_s * 3.6

        if best_end_time is None:
            best_index = len(times) - 1
        else:
            best_index = min(int(round(best_end_time / self.dt)), len(times) - 1)

        return -best_speed + self.penalties(realised, best_index)

    def run(self, n_jobs=0, verbose=1):
        """Run the optimisation.

        Args:
            n_jobs: number of worker processes. 0 runs serially.
            verbose: CMA-ES output level. 0 is silent.

        Returns:
            An OptimisationResult. Power after the fastest 200 m segment is
            not constrained by the penalties and should be ignored.
        """
        x0 = [self.model.cp] * self.n
        sigma0 = 10

        es = cma.CMAEvolutionStrategy(x0, sigma0, {
            'bounds': list(self.bounds),
            'maxiter': 1000,
            'popsize': 25,
            'tolx': 1e-6,
            'tolfun': 0.03,
            'verbose': verbose
        })
        es.optimize(self.evaluate, n_jobs=n_jobs)

        best_power = self.model.realise(es.result.xbest, self.dt)
        times = np.arange(0, self.duration, self.dt)

        run = Run.from_arrays(times, best_power, label="Optimised")
        run.calc_speeds = np.array(self.vehicle.calculate_speeds(times, best_power))

        t, d = self.distance_profile(best_power)
        speed = fastest_200m_segment(t, d)[0] * 3.6
        return OptimisationResult(run, speed)

import os
import matplotlib.pyplot as plt

SERIES_META = {
    'calc_speeds': dict(axis='speed', unit='km/h', label='Simulated Speed'),
    'speeds':      dict(axis='speed', unit='km/h', label='Measured Speed'),
    'watts':       dict(axis='power', unit='W', label='Power'),
}

DEFAULT_SERIES = ['calc_speeds', 'speeds', 'watts']

DATASET_STYLES = [
    {'calc_speeds': dict(color='orange', linestyle='-'),
     'speeds':      dict(color='red', linestyle='--', alpha=0.9),
     'watts':       dict(color='purple', linestyle='-')},
    {'calc_speeds': dict(color='green', linestyle='-'),
     'speeds':      dict(color='pink', linestyle='--', alpha=0.9),
     'watts':       dict(color='blue', linestyle='-')},
]

plt.rcParams.update({
    "text.usetex": False,
    "mathtext.fontset": "cm",
    "font.family": "serif",
    "font.serif": ["cmr10", "Computer Modern Roman", "DejaVu Serif"],
    "axes.formatter.use_mathtext": True,
    "axes.spines.top": False,
    "axes.grid": True,
    "grid.alpha": 0.3,
    "font.size": 27,
})

def _plot_series(ax1, ax2, run, series, style, labels, suffix):
    """Plot the requested series from one run. Returns the Line2D handles."""
    times = run.times
    lines = []

    for key in series:
        values = getattr(run, key, None)
        if values is None:
            continue
        if key not in SERIES_META:
            raise ValueError(f"Unknown series '{key}'. Known series: {list(SERIES_META)}")

        meta = SERIES_META[key]

        if key == 'watts':
            values = list(values)
            values[0] = 0.0

        x = times[:len(values)]
        y = values

        ax = ax1 if meta['axis'] == 'speed' else ax2
        label_text = labels.get(key) or meta['label'] + suffix
        label = f"{label_text} (Max: {max(y):.2f} {meta['unit']})"
        line, = ax.plot(x, y, label=label, **style.get(key, {}))
        lines.append(line)

    return lines


def plot(*runs, series=None, labels=None, show=True, savefig=None):
    """Plot one or more runs on shared axes, speed on the left and power on the right.

    Args:
        *runs: Run objects to plot.
        series: keys to plot from every run, or a list of lists, one per run.
            Keys are "watts", "speeds" and "calc_speeds". Defaults to all
            that each run has.
        labels: list of dicts, one per run, mapping series key to legend text.
        show: call plt.show().
        savefig: file path. A .pdf and a .png are both written.

    Returns:
        The left matplotlib Axes.

    Example:
        hpvopt.plot(optimised, measured,
                    series=[["watts"], ["watts", "calc_speeds"]],
                    labels=[{"watts": "Optimised Power"}, {"watts": "Measured Power"}])
    """

    series = list(DEFAULT_SERIES) if series is None else series
    if isinstance(series[0], str):
        series = [series] * len(runs)
    labels = labels or [{}] * len(runs)

    fig, ax1 = plt.subplots(figsize=(12, 8))
    ax1.set_xlabel('Time (s)')
    ax1.set_ylabel('Speed (km/h)')

    ax2 = ax1.twinx()
    ax2.set_ylabel('Power (W)')

    lines = []
    for i, run in enumerate(runs):
        suffix = f" ({run.label})" if len(runs) > 1 else ""
        lines += _plot_series(ax1, ax2, run, series[i],
                              DATASET_STYLES[i % len(DATASET_STYLES)], labels[i], suffix)

    ax1.set_zorder(ax2.get_zorder() + 1)
    ax1.patch.set_visible(False)

    ax1.legend(lines, [line.get_label() for line in lines],
               framealpha=.81, loc="lower right")
    ax1.grid(True, which='both', axis='both')
    ax1.minorticks_on()

    fig.tight_layout()
    if savefig:
        root = os.path.splitext(savefig)[0]
        plt.savefig(root + '.pdf')
        plt.savefig(root + '.png')

    if show:
        plt.show()
    return ax1

import os
import matplotlib.pyplot as plt
import numpy as np

from .models import MODELS

FITTED = ("cp", "exp", "biexp")  # the disproportionate model shares the cp fit

class Rider:
    """A rider's mean-maximal-power (MMP) curve and the models fitted to it.

    After `fit`, `models` holds the fitted physiological models:
        "cp": CPModel, critical power and W'.
        "exp": ExpModel, single-exponential power ceiling.
        "biexp": BiExpModel, bi-exponential power ceiling.
        "disproportionate": DisproportionateModel, shares the "cp" fit.
    Fitted values are in `model.params`.
    """
    def __init__(self, times, watts):
        """
        Args:
            times: MMP durations in s.
            watts: best average power in W for each duration.
        """
        order = np.argsort(times)
        self.times = np.asarray(times, float)[order]
        self.watts = np.asarray(watts, float)[order]
        self.models = {name: cls() for name, cls in MODELS.items()}

    def fit(self):
        """Fit every model to the MMP data.

        Returns:
            The Rider, so calls can be chained: `Rider(t, w).fit()`.
        """

        for model in self.models.values():
            model.fit(self.times, self.watts)
        return self

    def fit_quality(self):
        """Print and return how well each model fits the MMP data.

        Returns:
            dict of model name ("cp", "exp", "biexp") to a dict with `rss`,
            `rmse`, `mape`, `max_pct` and per-point `pct_err`.
        """
        t, actual = self.times, self.watts
        n = len(actual)

        print(f"{'Model':<12}{'RSS':>12}{'RMSE':>10}{'MAPE%':>9}{'maxErr%':>10}")
        results = {}
        for name in FITTED:
            resid = actual - self.models[name].predict(t)
            pct_err = np.abs(resid) / actual * 100

            rss = float(np.sum(resid ** 2))
            rmse = float(np.sqrt(rss / n))
            mape = float(np.mean(pct_err))
            max_pct = float(np.max(pct_err))

            results[name] = dict(rss=rss, rmse=rmse, mape=mape, max_pct=max_pct, pct_err=pct_err)
            print(f"{name:<12}{rss:>12.0f}{rmse:>10.1f}{mape:>9.2f}{max_pct:>10.2f}")
        return results

    def plot_fits(self, show=True, savefig=None):
        """Plot the MMP data with the fitted model curves.

        Args:
            show: call plt.show().
            savefig: file path to save a PDF to, or None.

        Returns:
            The matplotlib Axes.
        """
        plot_t = np.linspace(min(self.times), max(self.times), 500)
        colors = {"cp": "red", "exp": "blue", "biexp": "green"}

        fig, ax = plt.subplots(figsize=(8, 5))
        ax.scatter(self.times, self.watts, color="black", zorder=5, label="MMP data")
        for name in FITTED:
            model = self.models[name]
            t = plot_t[plot_t >= model.valid_from]
            ax.plot(t, model.predict(t), color=colors[name],
                    label=f"{name} (CP={model.cp:.0f}W)")
        ax.set_xlabel("Time (s)")
        ax.set_ylabel("Power (W)")
        ax.legend()
        fig.tight_layout()
        if savefig:
            plt.savefig(os.path.splitext(savefig)[0] + '.pdf')
        if show:
            plt.show()
        return ax

from . import plotting

NS = {
    'tcx': 'http://www.garmin.com/xmlschemas/TrainingCenterDatabase/v2',
    'ns3': 'http://www.garmin.com/xmlschemas/ActivityExtension/v2',
}

SERIES = ('times', 'distances', 'cadences', 'watts', 'speeds', 'calc_speeds')

def _text(elem, path, ns, cast=str):
    """Find a sub-element and cast its text, or return None if missing/empty."""
    found = elem.find(path, ns)
    if found is None or found.text is None:
        return None
    return cast(found.text)


def get_tcx_data(file_path):
    """
    Parses a TCX file to extract time, distance, cadence, and power data.
    """
    root = ET.parse(file_path).getroot()
    data = dict(times=[], distances=[], cadences=[], watts=[], speeds=[])

    for tp in root.findall('.//tcx:Trackpoint', NS):
        data['times'].append(_text(tp, 'tcx:Time', NS))
        data['distances'].append(_text(tp, 'tcx:DistanceMeters', NS, float) or 0)
        data['cadences'].append(_text(tp, 'tcx:Cadence', NS, int) or 0)

        watts_val = speed_val = None
        tpx = tp.find('tcx:Extensions/ns3:TPX', NS)
        if tpx is not None:
            watts_val = _text(tpx, 'ns3:Watts', NS, int)
            speed_val = _text(tpx, 'ns3:Speed', NS, float)
            if speed_val is not None:
                speed_val *= 3.6  # m/s -> km/h

        data['watts'].append(watts_val or 0)
        data['speeds'].append(speed_val or 0)

    if data['times'] and data['times'][0] is not None:
        t0 = datetime.fromisoformat(data['times'][0].replace('Z', '+00:00'))
        data['times'] = [
            (datetime.fromisoformat(t.replace('Z', '+00:00')) - t0).total_seconds()
            if t is not None else 0
            for t in data['times']
        ]
    return data


class Run:
    """Power and speed against time for one ride.

    Load from a .tcx or .npz file, or build from arrays with `Run.from_arrays`.
    Attributes are numpy arrays, or None if not available:
        times: seconds from the start of the run.
        watts: power in W.
        speeds: measured speed in km/h.
        calc_speeds: simulated speed in km/h, set by `Vehicle.simulate`.
        distances, cadences: from the TCX file only.
    """

    def __init__(self, path=None, start=None, end=None, label=None, rider_mass=None):
        """Load a run from a file.

        Args:
            path: .tcx or .npz file. If None, an empty Run is created.
            start: first time to keep, in s from the start of the file.
            end: last time to keep, in s from the start of the file.
            label: name used in plot legends. Defaults to the file name.
            rider_mass: rider mass in kg for this run. If set, `Vehicle.fit`
                uses it instead of the vehicle's own rider_mass.

        Raises:
            ValueError: if the file type is not .tcx or .npz, or no samples
                fall between `start` and `end`.
        """
        for key in SERIES:
            setattr(self, key, None)
        self.label = label or (os.path.basename(path) if path else "Run")
        self.rider_mass = rider_mass  # used by Vehicle.fit if set
        if path is None:
            return

        ext = os.path.splitext(path)[1].lower()
        if ext == '.tcx':
            data = get_tcx_data(path)
        elif ext == '.npz':
            data = dict(np.load(path))
        else:
            raise ValueError(f"Unsupported file type '{ext}'. Expected .tcx or .npz")
        for key, values in data.items():
            setattr(self, key, np.array(values))

        if start is not None or end is not None:
            self.__dict__.update(self.slice(start, end).__dict__)

    @classmethod
    def from_arrays(cls, times, watts, speeds=None, label="Run"):
        """Build a Run from arrays.

        Args:
            times: sample times in s.
            watts: power in W, same length as `times`.
            speeds: measured speed in km/h, optional.
            label: name used in plot legends.
        """
        run = cls(label=label)
        run.times = np.asarray(times, float)
        run.watts = np.asarray(watts, float)
        if speeds is not None:
            run.speeds = np.asarray(speeds, float)
        return run

    def slice(self, start=None, end=None):
        """Return a copy containing only the samples between `start` and `end`.

        Times in the copy are shifted to start at 0. The original is unchanged.

        Args:
            start: first time to keep, in s. Defaults to the start of the run.
            end: last time to keep, in s. Defaults to the end of the run.
        """
        start = self.times[0] if start is None else start
        end = self.times[-1] if end is None else end

        indices = [i for i, t in enumerate(self.times) if start <= t <= end]
        if not indices:
            raise ValueError(
                f"No valid samples between {start} and {end} s "
                f"Run covers {self.times[0]} to {self.times[-1]} s")

        sliced = copy.copy(self)
        for key in SERIES:
            values = getattr(self, key)
            if values is not None:
                setattr(sliced, key, np.array([values[i] for i in indices if i < len(values)]))

        sliced.times = sliced.times - sliced.times[0]
        return sliced

    def save(self, path):
        """Save all available series to a .npz file that `Run(path)` can load."""

        np.savez(path, **{k: getattr(self, k) for k in SERIES \
            if getattr(self, k) is not None})

    def plot(self, *others, **kwargs):
        """Plot this run, optionally overlaid with other runs.

        Args:
            *others: further Run objects to overlay.
            **kwargs: passed to `hpvopt.plot` (series, labels, show, savefig).

        Example:
            run.plot(sim, series=["watts", "speeds", "calc_speeds"])
        """
        return plotting.plot(self, *others, **kwargs)

import math, copy
import numpy as np
from dataclasses import dataclass, replace
from scipy.optimize import minimize

BOUNDS = {'crr': (0.001, 0.2), 'cda': (0.01, 0.9), 'air_density': (0.8, 2.0)}

def fastest_200m_segment(times, distances):
    """
    Find the fastest 200m segment of the ride, using linear interpolation
    between samples
    Returns:
        (avg_speed_m_s, start_time, end_time) for the fastest 200m segment,
    """
    SEGMENT = 200.0
    n = len(distances)
    best_avg_speed_m_s = 0.0
    best_start_time = None
    best_end_time = None
    j = 0
    for i in range(n):
        target = distances[i] + SEGMENT
        if j < i:
            j = i
        while j < n and distances[j] < target:
            j += 1
        if j >= n:
            break  # not enough remaining distance to complete a 200m segment

        # interpolate the time at which cumulative distance crosses `target`
        if j == i:
            t_end = times[j]
        else:
            d0, d1 = distances[j - 1], distances[j]
            t0, t1 = times[j - 1], times[j]
            frac = 0.0 if d1 <= d0 else (target - d0) / (d1 - d0)
            frac = min(max(frac, 0.0), 1.0)
            t_end = t0 + frac * (t1 - t0)

        segment_time = t_end - times[i]
        if segment_time > 0:
            avg_speed_m_s = SEGMENT / segment_time
            if avg_speed_m_s > best_avg_speed_m_s:
                best_avg_speed_m_s = avg_speed_m_s
                best_start_time = times[i]
                best_end_time = t_end

    return best_avg_speed_m_s, best_start_time, best_end_time

@dataclass
class Vehicle:
    """Power-balance model of the vehicle and rider.

    Args:
        rider_mass: kg.
        bike_mass: kg.
        cda: drag coefficient times frontal area, m^2.
        crr: rolling resistance coefficient.
        air_density: kg/m^3.
        slope: road gradient in m/m, negative is downhill.
        wind: m/s, positive is a tailwind.
        drive_train_efficiency: fraction of rider power reaching the wheels.
        wheel_mass: kg per wheel, used for rotational kinetic energy.
        number_of_wheels: number of wheels.
        gravity: m/s^2.
    """
    rider_mass: float               # kg
    bike_mass: float                # kg
    cda: float                      # m^2
    crr: float
    air_density: float = 1.1        # kg/m^3
    slope: float = 0.0              # m/m, negative is downhill
    wind: float = 0.0               # m/s, positive is tail wind
    drive_train_efficiency: float = 0.97
    wheel_mass: float = 2           # kg per wheel
    number_of_wheels: int = 3
    gravity: float = 9.81           # m/s^2

    def __str__(self):
        return (f"Vehicle\n"
                f"  mass          {self.rider_mass} + {self.bike_mass} kg (rider + bike)\n"
                f"  CdA           {self.cda:.4f} m^2\n"
                f"  Crr           {self.crr:.5f}\n"
                f"  air density   {self.air_density:.3f} kg/m^3\n"
                f"  slope         {self.slope:.4f}\n"
                f"  wind          {self.wind:.2f} m/s (+ = tailwind)\n"
                f"  efficiency    {self.drive_train_efficiency}")

    def drag_power(self, speed):
        """Calculate power lost to aerodynamic drag at a given speed in km/h."""
        relative_speed = speed / 3.6 - self.wind
        return 0.5 * self.air_density * self.cda * relative_speed ** 3

    def rolling_resistance_power(self, speed):
        """Calculate power lost to rolling resistance at a given speed in km/h."""
        total_mass = self.rider_mass + self.bike_mass
        return self.crr * total_mass * self.gravity * (speed / 3.6)

    def slope_power(self, speed):
        """Calculates power asisted by gravity on a slope at a given speed in km/h."""
        total_mass = self.rider_mass + self.bike_mass
        return total_mass * self.gravity * math.sin(math.atan(self.slope)) * (speed / 3.6)

    def total_power_consumed(self, speed):
        """Calculate total power consumed at a given speed in km/hr."""
        return self.drag_power(speed) + self.rolling_resistance_power(speed) + self.slope_power(speed)

    def current_speed_from_energy(self, kinetic_energy):
        """Calculates current speed in km/h from kinetic linear and rotational energy."""
        total_mass = self.rider_mass + self.bike_mass
        kinetic_energy_demon = (0.5 * total_mass + 0.25 * self.number_of_wheels * self.wheel_mass)
        return np.sqrt(kinetic_energy / kinetic_energy_demon) * 3.6

    def calculate_speeds(self, times, watts):
        """
        Calculates the speed of the vehicle in km/hr through all time in 'times' array
        Returns len(times) - 1 speeds, one per step between samples.
        """
        time_step = times[1] - times[0]
        current_speed = 0
        current_energy = 0
        calc_speeds = []
        for i in range(len(times) - 1):
            step_energy_produced = 0.5 * (watts[i] + watts[i+1]) * time_step * self.drive_train_efficiency
            step_energy_consumed = self.total_power_consumed(current_speed) * time_step
            current_energy += (step_energy_produced - step_energy_consumed)
            if current_energy < 0:
                current_energy = 0
            if i > 0 and (times[i] - times[i-1]) > time_step:
                current_energy = 0
            current_speed = self.current_speed_from_energy(current_energy)
            calc_speeds.append(current_speed)
        return calc_speeds

    def simulate(self, run):
        """Simulate the speed produced by the power in `run`.

        Args:
            run: Run with `times` and `watts`.

        Returns:
            A copy of `run` with `calc_speeds` set, in km/h. It has one fewer
            sample than `times`, since each value is the speed after a step.
            The original run is unchanged.
        """
        sim = copy.copy(run)
        sim.calc_speeds = np.array(self.calculate_speeds(run.times, run.watts))
        return sim

    def fit(self, runs, free=('cda', 'crr', 'air_density'), bounds=None):
        """Fit vehicle parameters to measured speed.

        Args:
            runs: list of Run objects with measured `speeds`.
            free: names of the parameters to fit.
            bounds: dict of name -> (low, high), overriding the defaults.

        Returns:
            A new Vehicle with the fitted parameters. The original is unchanged.

        Example:
            fitted = vehicle.fit([run1, run2], free=("cda", "crr"))
        """
        bounds = BOUNDS

        def objective(params):
            total_squared_error = 0
            total_count = 0
            for run in runs:
                vehicle = replace(self, **dict(zip(free, params)))
                if run.rider_mass is not None:
                    vehicle = replace(vehicle, rider_mass=run.rider_mass)
                sim_speeds = vehicle.calculate_speeds(run.times, run.watts)
                recorded_speeds = run.speeds[:-1]
                min_len = min(len(sim_speeds), len(recorded_speeds))
                error = np.sum((np.array(sim_speeds[:min_len]) -
                    np.array(recorded_speeds[:min_len])) ** 2)
                total_squared_error += error
                total_count += min_len
            return np.sqrt(total_squared_error / total_count)

        initial_guess = [getattr(self, k) for k in free]
        result = minimize(objective, initial_guess,
                          bounds=[bounds.get(k, (None, None)) for k in free],
                          method='L-BFGS-B')
        return replace(self, **dict(zip(free, result.x)))
